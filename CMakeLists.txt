cmake_minimum_required(VERSION 3.24)
project(RouteTec VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ROUTEMINE_ENABLE_CUDA "Build optional CUDA grade backend" OFF)
option(ROUTEMINE_BUILD_VIEWER "Build Qt/VTK native viewer" OFF)
include(CTest)
add_library(routemine_core src/geometry.cpp src/alignment.cpp src/adaptive_road.cpp src/design_review.cpp src/continuous_search.cpp src/road_candidates.cpp src/road_standards.cpp src/route_refinement.cpp src/earthwork_segments.cpp src/ramp_search.cpp src/validation.cpp src/backend_cpu.cpp src/cost.cpp src/obj.cpp src/dxf.cpp src/terrain_io.cpp src/route_export.cpp src/search.cpp src/road_search.cpp src/config.cpp)
target_include_directories(routemine_core PUBLIC include)
if(MSVC)
  target_compile_options(routemine_core PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4;/permissive-;/utf-8>)
else()
  target_compile_options(routemine_core PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic>)
endif()
if(ROUTEMINE_ENABLE_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  target_sources(routemine_core PRIVATE src/backend_cuda.cu)
  target_link_libraries(routemine_core PRIVATE CUDA::cudart)
  target_compile_definitions(routemine_core PUBLIC ROUTEMINE_HAS_CUDA=1)
  set_target_properties(routemine_core PROPERTIES CUDA_STANDARD 20 CUDA_STANDARD_REQUIRED YES)
endif()
add_executable(routemine_cli app/main.cpp app/plan_cli.cpp)
target_link_libraries(routemine_cli PRIVATE routemine_core)

if(ROUTEMINE_BUILD_VIEWER)
  find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets OpenGLWidgets Concurrent)
  find_package(VTK REQUIRED COMPONENTS CommonCore CommonDataModel FiltersSources RenderingCore RenderingOpenGL2 RenderingAnnotation InteractionStyle GUISupportQt)
  if(DEFINED VTK_QT_VERSION AND NOT VTK_QT_VERSION STREQUAL "6")
    message(FATAL_ERROR "RouteTec requires VTK built with Qt 6")
  endif()
  add_executable(routemine_viewer app/viewer.cpp app/terrain_view.cpp app/project_io.cpp)
  set_target_properties(routemine_viewer PROPERTIES OUTPUT_NAME "RouteTec")
  target_link_libraries(routemine_viewer PRIVATE routemine_core Qt6::Widgets Qt6::OpenGLWidgets Qt6::Concurrent ${VTK_LIBRARIES})
  if(WIN32)
    target_sources(routemine_viewer PRIVATE packaging/RouteTec.rc)
    find_package(Qt6 REQUIRED COMPONENTS EntryPointPrivate)
    set_target_properties(routemine_viewer PROPERTIES WIN32_EXECUTABLE TRUE)
    target_link_libraries(routemine_viewer PRIVATE Qt6::EntryPointPrivate)
    set_target_properties(routemine_viewer PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_BINARY_DIR}/Release/terrain-options")
  endif()
  if(MSVC)
    target_compile_options(routemine_viewer PRIVATE /W4 /permissive- /utf-8)
  endif()
  vtk_module_autoinit(TARGETS routemine_viewer MODULES ${VTK_LIBRARIES})
  if(BUILD_TESTING)
    add_executable(routemine_project_tests tests/project_tests.cpp app/project_io.cpp)
    target_include_directories(routemine_project_tests PRIVATE app)
    target_link_libraries(routemine_project_tests PRIVATE routemine_core Qt6::Core)
    if(MSVC)
      target_compile_options(routemine_project_tests PRIVATE /utf-8)
    endif()
    foreach(case roundtrip design_review_roundtrip design_review_invalid corridor_roundtrip legacy invalid atomic road_roundtrip alignment_roundtrip ramp_roundtrip local_roundtrip standards_review)
      add_test(NAME project_${case} COMMAND routemine_project_tests ${case})
    endforeach()
  endif()
endif()
