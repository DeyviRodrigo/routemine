cmake_minimum_required(VERSION 3.24)
project(RouteMine VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ROUTEMINE_ENABLE_CUDA "Build optional CUDA grade backend" OFF)
option(ROUTEMINE_BUILD_VIEWER "Build Qt/VTK native viewer" OFF)
include(CTest)
add_library(routemine_core src/geometry.cpp src/validation.cpp src/backend_cpu.cpp src/cost.cpp src/obj.cpp src/search.cpp src/config.cpp)
target_include_directories(routemine_core PUBLIC include)
if(MSVC)
  target_compile_options(routemine_core PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4;/permissive-;/utf-8>)
else()
  target_compile_options(routemine_core PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic>)
endif()
if(ROUTEMINE_ENABLE_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  target_sources(routemine_core PRIVATE src/backend_cuda.cu)
  target_link_libraries(routemine_core PRIVATE CUDA::cudart)
  target_compile_definitions(routemine_core PUBLIC ROUTEMINE_HAS_CUDA=1)
  set_target_properties(routemine_core PROPERTIES CUDA_STANDARD 20 CUDA_STANDARD_REQUIRED YES)
endif()
add_executable(routemine_cli app/main.cpp app/plan_cli.cpp)
target_link_libraries(routemine_cli PRIVATE routemine_core)
if(BUILD_TESTING)
  add_executable(routemine_tests tests/core_tests.cpp)
  target_link_libraries(routemine_tests PRIVATE routemine_core)
  foreach(case utm grade invalid mesh hole overlap cut_fill surface incomplete backend economics cost_balance)
    add_test(NAME ${case} COMMAND routemine_tests ${case})
  endforeach()
  add_executable(routemine_planner_tests tests/planner_tests.cpp)
  target_link_libraries(routemine_planner_tests PRIVATE routemine_core)
  foreach(case obj obj_invalid generated_economics generated_profile no_path search_limits discontinuity large_coordinates)
    add_test(NAME ${case} COMMAND routemine_planner_tests ${case})
  endforeach()
  add_test(NAME cli_plan COMMAND routemine_cli --plan ${CMAKE_CURRENT_SOURCE_DIR}/fixtures/ridge.obj ${CMAKE_CURRENT_SOURCE_DIR}/fixtures/ridge.request ${CMAKE_CURRENT_BINARY_DIR}/ridge-route.csv)
  add_executable(routemine_config_tests tests/config_tests.cpp)
  target_link_libraries(routemine_config_tests PRIVATE routemine_core)
  foreach(case defaults precedence validation persistence legacy)
    add_test(NAME config_${case} COMMAND routemine_config_tests ${case})
  endforeach()
  add_test(NAME cli_basic COMMAND routemine_cli --plan-basic ${CMAKE_CURRENT_SOURCE_DIR}/fixtures/ridge.obj ${CMAKE_CURRENT_SOURCE_DIR}/fixtures/basic.request ${CMAKE_CURRENT_BINARY_DIR}/basic-route.csv)
  add_test(NAME cli_show_config COMMAND routemine_cli --show-config ${CMAKE_CURRENT_SOURCE_DIR}/fixtures/ridge.obj ${CMAKE_CURRENT_SOURCE_DIR}/fixtures/basic.request)
endif()
if(ROUTEMINE_BUILD_VIEWER)
  find_package(Qt6 REQUIRED COMPONENTS Widgets OpenGLWidgets)
  find_package(VTK REQUIRED COMPONENTS CommonCore CommonDataModel FiltersSources RenderingCore RenderingOpenGL2 InteractionStyle GUISupportQt)
  add_executable(routemine_viewer app/viewer.cpp)
  target_link_libraries(routemine_viewer PRIVATE routemine_core Qt6::Widgets Qt6::OpenGLWidgets ${VTK_LIBRARIES})
  vtk_module_autoinit(TARGETS routemine_viewer MODULES ${VTK_LIBRARIES})
endif()
